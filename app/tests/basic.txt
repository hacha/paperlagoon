# PaperLagoon in the host simulator: ./build.sh --host, then
#   app/build/host/paperlagoon_host app/tests/basic.txt app/build/host
# It plays from the start; a minute of it goes to listen.wav (to hear, not only to check).
wait 600
expect playing 1
# the first notes, about 1.5 s after the start: their ripples on the screen
wait 1300
expect voices >0
shot ripples.png
peakreset
wav listen.wav
wait 20000
shot playing.png
wait 40000
wavstop
expect notes >10
expect peak_db_min -30
expect peak_db_max -0.5
# PLAY rests it; the tails ring on, then quiet
tap PLAY
wait 30
expect playing 0
wait 20000
expect voices 0
tap PLAY
wait 30
expect playing 1
# K1 turned: Density changes and SAVE keeps it over a restart
turn K1 5
wait 30
expect param0 55
expect dirty 1
tap SAVE
wait 30
expect dirty 0
reboot
wait 600
expect param0 55
